We set weakly informative priors. We truncate the intercept prior to include only positive values, with the lowest point to be 70 Hz. Any lower values would be erroneous octave jumps and they were discarded in the pre-processing.

get_prior(
  wrist_f0peak_acc ~ 1 +
    speech_rate_s +
    point_direct_s +
    wrist_apeak +
    sex_s +
    point_direct_s * wrist_apeak +
    (1 | speaker) +
    (0 + speech_rate_s + point_direct_s + wrist_apeak + point_direct_s * wrist_apeak | speaker) +
    (1 | rhyme) +
    (0 + speech_rate_s + point_direct_s + wrist_apeak + sex_s + point_direct_s * wrist_apeak | rhyme),
  data = f0_acc,
  family = "gaussian"
)

prior1 <- set_prior("truncated_normal(0, 200, lower = 70)", class = "Intercept")
prior2 <- set_prior("normal(0, 75)", class = "b", coef = "sex_s")
prior3 <- set_prior("normal(0, 10)", class = "b", coef = "point_direct_s")
prior4 <- set_prior("normal(0, 1)", class = "b", coef = "point_direct_s:wrist_apeak")
prior5 <- set_prior("normal(0, 10)", class = "b", coef = "speech_rate_s")
prior6 <- set_prior("normal(0, 1)", class = "b", coef = "wrist_apeak")
prior7 <- set_prior("normal(0, 100)", class = "sd")
prior8 <- set_prior("normal(0, 100)", class = "sigma")
prior9 <- set_prior("lkj(2)", class = "cor")
priors_f0_acc <- list(prior1, prior2, prior3, prior4, prior5, prior6, prior7, prior8, prior9)
priors_f0_acc <- prior_summary(priors_f0_acc)

Gaussian + priors

This is the model code.

f0peak_acc <- brm(
  wrist_f0peak_acc ~ 1 +
    speech_rate_s +
    point_direct_s +
    wrist_apeak +
    sex_s +
    point_direct_s * wrist_apeak +
    (1 | speaker) +
    (0 + speech_rate_s + point_direct_s + wrist_apeak + point_direct_s * wrist_apeak || speaker) +
    (1 | rhyme) +
    (0 + speech_rate_s + point_direct_s + wrist_apeak + sex_s + point_direct_s * wrist_apeak || rhyme),
  data = f0_acc,
  family = "gaussian",
  prior = priors_f0_acc,
  # Technical stuff
  backend = "cmdstanr",
  # Number of chains
  chains = 4, 
  # Number of iterations per chain
  iter = 10000,
  warmup = 2000,
  # Number of cores to use
  cores = 4,
  # Seed for replicability
  seed = 123,
  # Convergence
  control = list(adapt_delta = 0.99),
  # Save here
  file = paste0(parentfolder, "/9_statistical_analysis_scripts_and_model_output/f0peak_acc.rds"))
